CREATE TABLE public.categorias (
    id_categoria integer GENERATED ALWAYS AS IDENTITY NOT NULL,
    nombre_categoria character varying NOT NULL,
    CONSTRAINT categorias_pkey PRIMARY KEY (id_categoria)
);

CREATE TABLE public.clientes (
    id_cliente integer GENERATED ALWAYS AS IDENTITY NOT NULL,
    activo boolean,
    ciudad character varying,
    codigo_postal character varying,
    contrasena character varying NOT NULL,
    direccion character varying,
    email character varying NOT NULL UNIQUE,
    fecha_registro timestamp without time zone,
    nombre character varying NOT NULL,
    telefono character varying,
    comuna character varying,
    rut character varying,
    CONSTRAINT clientes_pkey PRIMARY KEY (id_cliente)
);

CREATE TABLE public.estados_pedido (
    id_estado integer GENERATED ALWAYS AS IDENTITY NOT NULL,
    nombre_estado character varying NOT NULL,
    CONSTRAINT estados_pedido_pkey PRIMARY KEY (id_estado)
);

CREATE TABLE public.proveedores (
    id_proveedor integer GENERATED ALWAYS AS IDENTITY NOT NULL,
    datos_contacto character varying NOT NULL,

    nombre_proveedor character varying NOT NULL,
    telefono character varying,
    CONSTRAINT proveedores_pkey PRIMARY KEY (id_proveedor)
);

CREATE TABLE public.trabajadores (
    id_trabajador integer GENERATED ALWAYS AS IDENTITY NOT NULL,
    activo boolean,
    contrasena character varying NOT NULL,
    email character varying NOT NULL UNIQUE,
    fecha_contratacion timestamp without time zone,
    fecha_registro timestamp without time zone,
    nombre character varying NOT NULL,
    rol character varying NOT NULL CHECK (
        rol::text = ANY (
            ARRAY[
                'ADMIN'::character varying,
                'CAJERO'::character varying,
                'BODEGUERO'::character varying,
                'VENDEDOR'::character varying,
                'CONTADOR'::character varying
            ]::text[]
        )
    ),
    telefono character varying,
    CONSTRAINT trabajadores_pkey PRIMARY KEY (id_trabajador)
);

INSERT INTO public.estados_pedido (nombre_estado)
SELECT estado
FROM (VALUES
    ('pendiente'),
    ('pagado'),
    ('preparacion'),
    ('despachado'),
    ('entregado')
) AS seed(estado)
WHERE NOT EXISTS (
    SELECT 1
    FROM public.estados_pedido existente
    WHERE lower(existente.nombre_estado) = lower(seed.estado)
);
