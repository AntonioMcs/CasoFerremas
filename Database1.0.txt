-- ==========================================
-- TABLA: categorias
-- ==========================================

CREATE TABLE categorias (
    id_categoria SERIAL PRIMARY KEY,
    nombre_categoria VARCHAR(100) NOT NULL
);

-- ==========================================
-- TABLA: proveedores
-- ==========================================

CREATE TABLE proveedores (
    id_proveedor SERIAL PRIMARY KEY,
    nombre_proveedor VARCHAR(100) NOT NULL,
    direccion VARCHAR(200),
    email VARCHAR(100),
    telefono VARCHAR(20),
    datos_contacto VARCHAR(255)
);

-- ==========================================
-- TABLA: usuarios
-- ==========================================

CREATE TABLE usuarios (
    id_usuario SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    contrasena VARCHAR(255) NOT NULL,
    tipo_usuario VARCHAR(30) NOT NULL,
    fecha_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ==========================================
-- TABLA: estados_pedido
-- ==========================================

CREATE TABLE estados_pedido (
    id_estado SERIAL PRIMARY KEY,
    nombre_estado VARCHAR(50) NOT NULL
);

-- ==========================================
-- TABLA: productos
-- ==========================================

CREATE TABLE productos (
    id_producto SERIAL PRIMARY KEY,
    codigo_sku VARCHAR(50) UNIQUE NOT NULL,
    nombre_producto VARCHAR(150) NOT NULL,
    descripcion VARCHAR(500),
    marca VARCHAR(100),
    precio NUMERIC(10,2) NOT NULL CHECK (precio >= 0),
    stock INTEGER NOT NULL DEFAULT 0 CHECK (stock >= 0),
    unidad_medida VARCHAR(50),
    fecha_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    id_categoria INTEGER NOT NULL,
    id_proveedor INTEGER NOT NULL,

    CONSTRAINT fk_producto_categoria
        FOREIGN KEY (id_categoria)
        REFERENCES categorias(id_categoria),

    CONSTRAINT fk_producto_proveedor
        FOREIGN KEY (id_proveedor)
        REFERENCES proveedores(id_proveedor)
);

-- ==========================================
-- TABLA: inventario
-- ==========================================

CREATE TABLE inventario (
    id_inventario SERIAL PRIMARY KEY,
    stock_actual INTEGER NOT NULL DEFAULT 0 CHECK (stock_actual >= 0),
    stock_minimo INTEGER NOT NULL DEFAULT 0 CHECK (stock_minimo >= 0),
    ubicacion_bodega VARCHAR(100),

    id_producto INTEGER NOT NULL,
    id_proveedor INTEGER NOT NULL,

    CONSTRAINT fk_inventario_producto
        FOREIGN KEY (id_producto)
        REFERENCES productos(id_producto),

    CONSTRAINT fk_inventario_proveedor
        FOREIGN KEY (id_proveedor)
        REFERENCES proveedores(id_proveedor)
);

-- ==========================================
-- TABLA: pedidos
-- ==========================================

CREATE TABLE pedidos (
    id_pedido SERIAL PRIMARY KEY,
    fecha_pedido TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    metodo_pago VARCHAR(50),
    tipo_entrega VARCHAR(50),
    total NUMERIC(12,2) NOT NULL CHECK (total >= 0),

    id_estado INTEGER NOT NULL,
    id_usuario INTEGER NOT NULL,

    CONSTRAINT fk_pedido_estado
        FOREIGN KEY (id_estado)
        REFERENCES estados_pedido(id_estado),

    CONSTRAINT fk_pedido_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuarios(id_usuario)
);

-- ==========================================
-- TABLA: detalle_pedido
-- ==========================================

CREATE TABLE detalle_pedido (
    id_detalle SERIAL PRIMARY KEY,
    cantidad INTEGER NOT NULL CHECK (cantidad > 0),
    precio_unitario NUMERIC(10,2) NOT NULL CHECK (precio_unitario >= 0),
    subtotal NUMERIC(12,2) NOT NULL CHECK (subtotal >= 0),

    id_pedido INTEGER NOT NULL,
    id_producto INTEGER NOT NULL,

    CONSTRAINT fk_detalle_pedido
        FOREIGN KEY (id_pedido)
        REFERENCES pedidos(id_pedido),

    CONSTRAINT fk_detalle_producto
        FOREIGN KEY (id_producto)
        REFERENCES productos(id_producto)
);

-- ==========================================
-- TABLA: carrito_compras
-- ==========================================

CREATE TABLE carrito_compras (
    id_carrito SERIAL PRIMARY KEY,
    cantidad INTEGER NOT NULL CHECK (cantidad > 0),
    descuento NUMERIC(10,2) DEFAULT 0 CHECK (descuento >= 0),
    precio_final NUMERIC(12,2) NOT NULL CHECK (precio_final >= 0),
    fecha_agregado TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    id_producto INTEGER NOT NULL,
    id_usuario INTEGER NOT NULL,

    CONSTRAINT fk_carrito_producto
        FOREIGN KEY (id_producto)
        REFERENCES productos(id_producto),

    CONSTRAINT fk_carrito_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuarios(id_usuario)
);

-- ==========================================
-- TABLA: facturacion_cliente
-- ==========================================

CREATE TABLE facturacion_cliente (
    id_facturacion SERIAL PRIMARY KEY,
    pedidos VARCHAR(100),
    comuna VARCHAR(100),
    direccion VARCHAR(200),
    nombre VARCHAR(100),
    rut VARCHAR(20),
    telefono VARCHAR(20),

    id_usuario INTEGER NOT NULL,

    CONSTRAINT fk_facturacion_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuarios(id_usuario)
);

-- ==========================================
-- TABLA: despachos
-- ==========================================

CREATE TABLE despachos (
    id_despacho SERIAL PRIMARY KEY,
    direccion_entrega VARCHAR(200) NOT NULL,
    estado_despacho VARCHAR(50) NOT NULL,
    fecha_entrega TIMESTAMP,
    fecha_envio TIMESTAMP,

    id_pedido INTEGER NOT NULL,

    CONSTRAINT fk_despacho_pedido
        FOREIGN KEY (id_pedido)
        REFERENCES pedidos(id_pedido)
);

-- ==========================================
-- TABLA: pagos
-- ==========================================

CREATE TABLE pagos (
    id_pago SERIAL PRIMARY KEY,
    estado_pago VARCHAR(50) NOT NULL,
    fecha_pago TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    metodo_pago VARCHAR(50),
    monto NUMERIC(12,2) NOT NULL CHECK (monto >= 0),

    id_pedido INTEGER NOT NULL,

    CONSTRAINT fk_pago_pedido
        FOREIGN KEY (id_pedido)
        REFERENCES pedidos(id_pedido)
);
